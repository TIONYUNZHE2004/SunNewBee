# UI Polish Implementation Plan (Project 4)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Give TownshipLens a real daylight skin (shared by Day and Dusk), an immersive full-bleed map with a floating glass assistant, answers that read at a glance without hiding any honesty note, polished map overlays and Tickets, and a phone bottom sheet.

**Architecture:** Tokens stay CSS variables in `apps/web/src/index.css`, keyed by `data-skin` (set by Project 3's `applyTheme`). Layout moves from a two-column grid to a full-bleed map with absolutely positioned glass panels; the camera keeps its subject centred in the visible map area through a frustum offset. One small backend addition (server-attached route summary fields on `route.highlight`) feeds the answer's summary strip, so the UI never computes a fact.

**Tech Stack:** React 19, zustand, Tailwind 4, Cesium 1.146 (frustum offset only), vitest 5 + Testing Library (jsdom per file), Playwright 1.64; Python 3.13 / PydanticAI 2.56 agent (one events field set), FastAPI 0.143 (OpenAPI regen).

**Spec:** `docs/superpowers/specs/2026-10-16-ui-polish-design.md` (and the brief inside `TownshipLens Frontend Redesign-handoff.zip`, `project/uploads/frontend-redesign-brief.md`, whose wording and accessibility rules still apply).

**Prerequisite:** Project 3 Phase 1 merged (`Theme`, `skinFor`, `data-skin`, `ThemeMenu`, `CameraControls`).

## Global Constraints

- Honesty notes (not-verified note, flood disclaimer with official links, demo-value labels) are never inside a collapsible and never hidden while their answer is shown. The collapsed evidence summary states the demo-value count.

```ts
// viewport.ts
export const PANEL_INSET_PX = 16;

export const panelWidthPx = (viewportW: number) => Math.round(Math.min(440, Math.max(360, viewportW * 0.3)));

export function visibleViewport(widthPx: number, panelPx: number, open: boolean) {
  const right = open ? widthPx - panelPx - PANEL_INSET_PX * 2 : widthPx;
  return { left: 0, right, centerX: right / 2 };
}

/**
 * PerspectiveFrustum.xOffset that moves the image so the camera's look point lands `shiftPx`
 * pixels LEFT of the canvas centre (the centre of the visible area when the panel is open).
 * Cesium's fov is horizontal when width > height, else vertical.
 */
export function frustumXOffset(shiftPx: number, widthPx: number, near: number, fovRad: number, heightPx: number): number {
  if (shiftPx === 0 || widthPx <= 0) return 0;
  const aspect = widthPx / Math.max(1, heightPx);
  const halfW = aspect >= 1 ? near * Math.tan(fovRad / 2) : near * Math.tan(fovRad / 2) * aspect;
  return (shiftPx / (widthPx / 2)) * halfW;
}
```

```ts
// panelStore.ts
import { create } from "zustand";

const KEY = "tl.panel.open";
const read = () => {
  try {
    return typeof localStorage === "undefined" ? null : localStorage.getItem(KEY);
  } catch {
    return null;
  }
};

interface PanelState {
  open: boolean;
  setOpen: (v: boolean) => void;
  sheet: "peek" | "half" | "full";
  setSheet: (s: "peek" | "half" | "full") => void;
}

export const usePanel = create<PanelState>((set) => ({
  open: read() !== "false",
  setOpen: (open) => {
    try {
      localStorage.setItem(KEY, String(open));
    } catch {
      /* per-viewer convenience only */
    }
    set({ open });
  },
  sheet: "peek",
  setSheet: (sheet) => set({ sheet }),
}));
```

```tsx
const useWide = () => {
  const q = "(min-width: 1024px)";
  const [wide, setWide] = useState(() => typeof matchMedia === "function" && matchMedia(q).matches);
  useEffect(() => {
    if (typeof matchMedia !== "function") return;
    const m = matchMedia(q);
    const on = () => setWide(m.matches);
    m.addEventListener("change", on);
    return () => m.removeEventListener("change", on);
  }, []);
  return wide;
};
```

```ts
export function reduceIntent(m: AssistantMessage, i: SceneIntent): AssistantMessage {
  if (i.kind === "route.highlight" && i.route_id)

      ...m,
      route: {
        id: i.route_id,
        status: i.route_status ?? undefined,
        distanceM: i.distance_m ?? undefined,
        minutes: i.est_minutes ?? undefined,
      },
    };
  if (i.kind === "flood.set_level" && i.scenario_id) return { ...m, scenarioId: i.scenario_id };
  return m;
}
```

```tsx
import { type ReactNode, useRef } from "react";
import { Icon } from "../../ui/icons";
import { usePanel } from "./panelStore";

const ORDER = ["peek", "half", "full"] as const;
type Sheet = (typeof ORDER)[number];
export const SHEET_HEIGHTS: Record<Sheet, string> = {
  peek: "9.5rem",
  half: "55dvh",
  full: "calc(100dvh - 5rem)",
};

export function BottomSheet({ children }: { children: ReactNode }) {
  const { sheet, setSheet } = usePanel();
  const i = ORDER.indexOf(sheet);
  const up = () => setSheet(ORDER[Math.min(2, i + 1)]!);
  const down = () => setSheet(ORDER[Math.max(0, i - 1)]!);
  const drag = useRef<{ y: number; i: number } | null>(null);
  return (
    <div
      className="glass absolute inset-x-0 bottom-0 z-20 flex flex-col rounded-t-3xl transition-[height] duration-200 ease-[cubic-bezier(.2,.7,.2,1)]"
      style={{ height: SHEET_HEIGHTS[sheet] }}
    >
      <div className="flex items-center justify-between px-3 pt-2">
        <button type="button" aria-label="Collapse assistant" onClick={down} disabled={i === 0}
          className="grid size-11 place-items-center rounded-full text-fg disabled:opacity-30">
          <Icon name="chevron" className="size-5 rotate-90" />
        </button>
        {/* biome-ignore lint/a11y/useSemanticElements: a draggable handle exposed as a slider */}
        <div
          role="slider"
          tabIndex={0}
          aria-label="Assistant height"
          aria-valuemin={0}
          aria-valuemax={2}
          aria-valuenow={i}
          aria-valuetext={sheet}
          onKeyDown={(e) => {
            if (e.key === "ArrowUp") { e.preventDefault(); up(); }
            if (e.key === "ArrowDown") { e.preventDefault(); down(); }
          }}
          onPointerDown={(e) => {
            drag.current = { y: e.clientY, i };
            (e.target as HTMLElement).setPointerCapture?.(e.pointerId);
          }}
          onPointerUp={(e) => {
            const d = drag.current;
            drag.current = null;
            if (!d) return;
            const dy = d.y - e.clientY;
            if (dy > 40) setSheet(ORDER[Math.min(2, d.i + 1)]!);
            else if (dy < -40) setSheet(ORDER[Math.max(0, d.i - 1)]!);
          }}
          className="h-6 w-16 cursor-grab touch-none rounded-full py-2.5"
        >
          <span className="block h-1.5 w-full rounded-full bg-fg/30" />
        </div>
        <button type="button" aria-label="Expand assistant" onClick={up} disabled={i === 2}
          className="grid size-11 place-items-center rounded-full text-fg disabled:opacity-30">
```
